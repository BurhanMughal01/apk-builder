let state={theme:'dark',fontSize:'medium',version:'hanafi'};
function init(){loadState();applyTheme();applyFontSize();render();}
function loadState(){try{const s=localStorage.getItem('qunoot_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.fontSize=p.fontSize||'medium';}}catch(e){}}
function saveState(){try{localStorage.setItem('qunoot_v1',JSON.stringify({theme:state.theme,fontSize:state.fontSize}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const i=document.getElementById('ti');if(i){if(state.theme==='dark'){i.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{i.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
function applyFontSize(){const s={small:'20px',medium:'24px',large:'32px'};document.documentElement.style.setProperty('--f-a',s[state.fontSize]);document.querySelectorAll('.fb').forEach(b=>{b.classList.toggle('active',b.dataset.size===state.fontSize);});}
function sF(s){state.fontSize=s;applyFontSize();saveState();}
function sV(v){state.version=v;document.getElementById('tH').classList.toggle('active',v==='hanafi');document.getElementById('tS').classList.toggle('active',v==='shafi');render();}
function render(){const d=DATA[state.version];const el=document.getElementById('qc');el.innerHTML='<div class="dc"><div class="dl">Arabic</div><div class="da">'+d.ar+'</div><div class="dt">'+d.tr+'</div><div class="du">'+d.ur+'</div></div>';}
function toggleAudio(){}
function tAudio(){const btn=document.getElementById('abtn');const stat=document.getElementById('astat');const isPlaying=btn.classList.contains('playing');if(isPlaying){btn.classList.remove('playing');document.getElementById('aicon').innerHTML='<polygon points="5 3 19 12 5 21 5 3"/>';stat.textContent='Tap to play';return;}const url='https://cdn.islamic.network/quran/audio-surah/128/ar.alafasy/1.mp3';const a=new Audio(url);a.play().then(()=>{btn.classList.add('playing');document.getElementById('aicon').innerHTML='<rect x="6" y="4" width="4" height="16"/><rect x="14" y="4" width="4" height="16"/>';stat.textContent='Playing... Tap to stop';a.onended=()=>{btn.classList.remove('playing');document.getElementById('aicon').innerHTML='<polygon points="5 3 19 12 5 21 5 3"/>';stat.textContent='Tap to play';};}).catch(()=>{stat.textContent='Audio not available';});}
function cD(){const d=DATA[state.version];const t=d.ar+'\n\n'+d.tr+'\n\n'+d.ur;if(navigator.clipboard){navigator.clipboard.writeText(t).then(()=>showToast('Copied'));}}
function sD(){const d=DATA[state.version];const t=d.ar+'\n\n'+d.tr+'\n\n'+d.ur;if(navigator.share){navigator.share({text:t}).catch(()=>{});}else cD();}
function sW(v){document.querySelectorAll('.view').forEach(x=>x.classList.remove('active'));document.getElementById('v-'+v).classList.add('active');document.querySelectorAll('.ni').forEach(b=>b.classList.toggle('active',b.dataset.v===v));}
let tt;function showToast(m){const t=document.getElementById('toast');document.getElementById('tt').textContent=m;t.classList.add('show');clearTimeout(tt);tt=setTimeout(()=>t.classList.remove('show'),1800);}
