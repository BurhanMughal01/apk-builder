let state={theme:'dark',fontSize:'medium',mode:'morning',morningCounts:{},eveningCounts:{},stats:{today:0,total:0,sessions:0,streak:0,lastDate:null}};
function init(){loadState();applyTheme();applyFontSize();renderAdhkar();updateStats();}
function loadState(){try{const s=localStorage.getItem('adhkar_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.fontSize=p.fontSize||'medium';state.morningCounts=p.morningCounts||{};state.eveningCounts=p.eveningCounts||{};state.stats=p.stats||state.stats;const today=new Date().toDateString();if(state.stats.lastDate!==today){if(state.stats.lastDate){const last=new Date(state.stats.lastDate);const now=new Date();const diff=Math.floor((now-last)/(1000*60*60*24));if(diff>1)state.stats.streak=0;}state.stats.today=0;state.stats.lastDate=today;}}}catch(e){}}
function saveState(){try{localStorage.setItem('adhkar_v1',JSON.stringify({theme:state.theme,fontSize:state.fontSize,morningCounts:state.morningCounts,eveningCounts:state.eveningCounts,stats:state.stats}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
function applyFontSize(){const sizes={small:'22px',medium:'26px',large:'32px'};document.documentElement.style.setProperty('--font-arabic',sizes[state.fontSize]);document.querySelectorAll('.font-btn').forEach(b=>{b.classList.toggle('active',b.dataset.size===state.fontSize);});}
function setFontSize(size){state.fontSize=size;applyFontSize();saveState();}
function switchTab(mode){state.mode=mode;document.getElementById('tabMorning').classList.toggle('active',mode==='morning');document.getElementById('tabEvening').classList.toggle('active',mode==='evening');renderAdhkar();}
function getCurrentList(){return state.mode==='morning'?MORNING_ADHKAR:EVENING_ADHKAR;}
function getCurrentCounts(){return state.mode==='morning'?state.morningCounts:state.eveningCounts;}
function renderAdhkar(){const list=getCurrentList();const counts=getCurrentCounts();const container=document.getElementById('adhkarList');container.innerHTML=list.map(a=>{const c=counts[a.id]||0;const done=c>=a.count;return '<div class="adhkar-card'+(done?' completed':'')+'"><div class="adhkar-top"><div class="adhkar-title">'+a.title+'</div><span class="adhkar-count-badge">×'+a.count+'</span></div><div class="adhkar-arabic">'+a.ar+'</div><div class="adhkar-trans">'+a.trans+'</div><div class="adhkar-ref">📚 '+a.ref+'</div><div class="counter-row"><button class="counter-btn" onclick="increment('+a.id+')" '+(done?'disabled':'')+'><svg viewBox="0 0 24 24"><path d="M12 5v14M5 12h14"/></svg>'+(done?'Completed':'Count')+'</button><div class="counter-display">'+c+'/'+a.count+'</div><button class="reset-mini" onclick="resetOne('+a.id+')"><svg viewBox="0 0 24 24"><path d="M3 12a9 9 0 109-9 9.75 9.75 0 00-6.74 2.74L3 8"/><path d="M3 3v5h5"/></svg></button></div></div>';}).join('');updateProgress();}
function increment(id){const list=getCurrentList();const item=list.find(a=>a.id===id);if(!item)return;const counts=getCurrentCounts();const c=counts[id]||0;if(c>=item.count)return;counts[id]=c+1;state.stats.today++;state.stats.total++;if(navigator.vibrate)navigator.vibrate(20);saveState();renderAdhkar();updateStats();if(counts[id]===item.count&&isAllComplete()){showToast('MashaAllah! All adhkar complete');state.stats.sessions++;if(state.stats.streak===0||state.stats.lastDate!==new Date().toDateString()){state.stats.streak++;state.stats.lastDate=new Date().toDateString();}saveState();updateStats();}}
function resetOne(id){const counts=getCurrentCounts();counts[id]=0;saveState();renderAdhkar();updateStats();}
function isAllComplete(){const list=getCurrentList();const counts=getCurrentCounts();return list.every(a=>(counts[a.id]||0)>=a.count);}
function updateProgress(){const list=getCurrentList();const counts=getCurrentCounts();const total=list.length;const done=list.filter(a=>(counts[a.id]||0)>=a.count).length;document.getElementById('progressCount').textContent=done+' / '+total;document.getElementById('progressFill').style.width=(done/total*100)+'%';document.getElementById('progressTitle').textContent=(state.mode==='morning'?'Morning':'Evening')+' Progress';document.getElementById('completeBanner').classList.toggle('show',done===total&&total>0);}
function updateStats(){document.getElementById('statToday').textContent=state.stats.today||0;document.getElementById('statTotal').textContent=(state.stats.total||0).toLocaleString();document.getElementById('statStreak').textContent=state.stats.streak||0;document.getElementById('statSessions').textContent=state.stats.sessions||0;}
function resetAll(){if(!confirm('Reset all progress? Cannot be undone.'))return;state.morningCounts={};state.eveningCounts={};state.stats={today:0,total:0,sessions:0,streak:0,lastDate:new Date().toDateString()};saveState();renderAdhkar();updateStats();showToast('All progress reset');}
function switchView(view){document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));document.getElementById('view-'+view).classList.add('active');document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});const titles={adhkar:'Adhkar',stats:'Progress',settings:'Settings'};document.getElementById('headerTitle').textContent=titles[view]||'Adhkar';document.getElementById('mainArea').scrollTo({top:0});}
let toastTimer;function showToast(msg){const t=document.getElementById('toast');document.getElementById('toastText').textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),2000);}
init();
