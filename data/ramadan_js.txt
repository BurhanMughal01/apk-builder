let state={theme:'dark',lat:24.8607,lng:67.0011,city:'Karachi',country:'Pakistan',tracker:{},ramadanStart:null,ramadanDay:null,sehri:null,iftar:null,hijriMonth:null,hijriYear:null,timer:null};
function init(){loadState();applyTheme();loadRamadanInfo();}
function loadState(){try{const s=localStorage.getItem('ramadan_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.lat=p.lat||24.8607;state.lng=p.lng||67.0011;state.city=p.city||'Karachi';state.country=p.country||'Pakistan';state.tracker=p.tracker||{};}}catch(e){}updateCityUI();}
function saveState(){try{localStorage.setItem('ramadan_v1',JSON.stringify({theme:state.theme,lat:state.lat,lng:state.lng,city:state.city,country:state.country,tracker:state.tracker}));}catch(e){}}
function updateCityUI(){const el=document.getElementById('citySub');if(el)el.textContent=state.city+', '+state.country;}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}

async function loadRamadanInfo(){
  const now=new Date();
  const dd=String(now.getDate()).padStart(2,'0');
  const mm=String(now.getMonth()+1).padStart(2,'0');
  const yyyy=now.getFullYear();
  try{
    const url='https://api.aladhan.com/v1/gToH/'+dd+'-'+mm+'-'+yyyy;
    const res=await fetch(url);
    const data=await res.json();
    if(data.data&&data.data.hijri){
      const h=data.data.hijri;
      state.hijriMonth=parseInt(h.month.number);
      state.hijriYear=parseInt(h.year);
      state.ramadanDay=parseInt(h.day);
    }
  }catch(e){console.log(e);}
  await loadTimes();
  renderTracker();
}

async function loadTimes(){
  const now=new Date();
  const dd=String(now.getDate()).padStart(2,'0');
  const mm=String(now.getMonth()+1).padStart(2,'0');
  const yyyy=now.getFullYear();
  const dateStr=dd+'-'+mm+'-'+yyyy;
  try{
    const url='https://api.aladhan.com/v1/timings/'+dateStr+'?latitude='+state.lat+'&longitude='+state.lng+'&method=1';
    const res=await fetch(url);
    const data=await res.json();
    if(data.data&&data.data.timings){
      state.sehri=data.data.timings.Fajr;
      state.iftar=data.data.timings.Maghrib;
    }
  }catch(e){state.sehri='04:30';state.iftar='18:15';}
}

function renderTracker(){
  const now=new Date();
  const isRamadan=state.hijriMonth===9;
  const ramadanDay=isRamadan?state.ramadanDay:null;
  let nextRamadanDays=0;
  if(!isRamadan){
    let year=state.hijriYear;
    if(state.hijriMonth>=9)year++;
    nextRamadanDays=Math.max(0,(9-state.hijriMonth)*30);
  }
  let html='';
  if(isRamadan){
    html+='<div class="countdown-hero"><div class="hero-label">Ramadan Day</div><div class="hero-number">'+ramadanDay+'</div><div class="hero-sub">of 30 days</div><div class="hero-hijri">🌙 '+state.hijriYear+' AH</div></div>';
  }else{
    html+='<div class="countdown-hero"><div class="hero-label">Ramadan Countdown</div><div class="hero-number">~'+nextRamadanDays+'</div><div class="hero-sub">days remaining</div><div class="hero-hijri">🌙 Prepare for Ramadan</div></div>';
  }
  html+='<div class="section-title"><svg viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M12 6v6l4 2"/></svg>Today\\'s Timings</div>';
  html+='<div class="time-grid"><div class="time-card sehri"><div class="time-icon">🌌</div><div class="time-label">Sehri Ends</div><div class="time-value">'+(state.sehri||'--:--')+'</div><div class="time-sub">Fajr time</div></div><div class="time-card iftar"><div class="time-icon">🌅</div><div class="time-label">Iftar Time</div><div class="time-value">'+(state.iftar||'--:--')+'</div><div class="time-sub">Maghrib time</div></div></div>';
  html+='<div class="section-title"><svg viewBox="0 0 24 24"><path d="M9 11l3 3L22 4"/><path d="M21 12v7a2 2 0 01-2 2H5a2 2 0 01-2-2V5a2 2 0 012-2h11"/></svg>30-Day Tracker</div>';
  const totalDone=Object.values(state.tracker).filter(v=>v).length;
  const percent=Math.round((totalDone/30)*100);
  html+='<div class="progress-card"><div class="progress-head"><span class="progress-text">Fasts Completed</span><span class="progress-count">'+totalDone+' / 30</span></div><div class="progress-bar"><div class="progress-fill" style="width:'+percent+'%"></div></div><div class="progress-stats"><div class="progress-stat"><div class="progress-stat-val">'+totalDone+'</div><div class="progress-stat-label">Done</div></div><div class="progress-stat"><div class="progress-stat-val">'+(30-totalDone)+'</div><div class="progress-stat-label">Remaining</div></div><div class="progress-stat"><div class="progress-stat-val">'+percent+'%</div><div class="progress-stat-label">Progress</div></div></div></div>';
  html+='<div class="days-grid">';
  for(let i=1;i<=30;i++){
    const done=state.tracker[i];
    const isToday=ramadanDay===i;
    const isPast=ramadanDay&&i<ramadanDay;
    let cls='day-cell';
    if(done)cls+=' done';
    if(isToday)cls+=' today';
    if(isPast&&!done)cls+=' past';
    html+='<div class="'+cls+'" onclick="toggleDay('+i+')">'+i+'</div>';
  }
  html+='</div>';
  html+='<div class="daily-card"><div class="daily-title">✨ Ramadan Dua</div><div class="daily-arabic">اللَّهُمَّ إِنَّكَ عَفُوٌّ تُحِبُّ الْعَفْوَ فَاعْفُ عَنِّي</div><div class="daily-trans">"O Allah, You are Most Forgiving, and You love forgiveness; so forgive me."<br><span style="color:var(--text-subtle);font-size:11px">— Tirmidhi 3513</span></div></div>';
  document.getElementById('trackerContent').innerHTML=html;
}

function toggleDay(day){
  state.tracker[day]=!state.tracker[day];
  if(state.tracker[day]&&navigator.vibrate)navigator.vibrate(30);
  saveState();
  renderTracker();
  if(state.tracker[day])showToast('Day '+day+' marked ✓');
}

async function saveCity(){
  const input=document.getElementById('cityInput');
  const city=input.value.trim();
  if(!city){showToast('Enter a city name');return;}
  showToast('Searching...');
  try{
    const url='https://nominatim.openstreetmap.org/search?format=json&q='+encodeURIComponent(city)+'&limit=1';
    const res=await fetch(url,{headers:{'Accept':'application/json'}});
    const data=await res.json();
    if(!data||data.length===0){showToast('City not found');return;}
    const place=data[0];
    state.city=place.display_name.split(',')[0];
    state.country=place.display_name.split(',').slice(-1)[0].trim();
    state.lat=parseFloat(place.lat);
    state.lng=parseFloat(place.lon);
    saveState();
    updateCityUI();
    input.value='';
    showToast('City updated');
    await loadTimes();
    renderTracker();
  }catch(e){showToast('Search failed');}
}

function resetAll(){
  if(!confirm('Reset all 30 days tracker?'))return;
  state.tracker={};
  saveState();
  renderTracker();
  showToast('Progress reset');
}

function switchView(view){
  document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));
  document.getElementById('view-'+view).classList.add('active');
  document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});
  const titles={tracker:'Ramadan Tracker',settings:'Settings'};
  document.getElementById('headerTitle').textContent=titles[view]||'Ramadan Tracker';
  document.getElementById('mainArea').scrollTo({top:0});
}
