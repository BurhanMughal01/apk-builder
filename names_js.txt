let state={theme:'dark',currentCat:'all',favorites:[],current:null};
function init(){loadState();applyTheme();renderCategories();renderNames();renderFavs();}
function loadState(){try{const s=localStorage.getItem('names_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.favorites=p.favorites||[];}}catch(e){}}
function saveState(){try{localStorage.setItem('names_v1',JSON.stringify({theme:state.theme,favorites:state.favorites}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
function renderCategories(){const el=document.getElementById('catChips');el.innerHTML=CATEGORIES.map(c=>'<button class="chip'+(c.id===state.currentCat?' active':'')+'" onclick="setCategory(\''+c.id+'\')">'+c.icon+' '+c.name+'</button>').join('');}
function setCategory(id){state.currentCat=id;renderCategories();renderNames();document.getElementById('mainArea').scrollTo({top:0,behavior:'smooth'});}
function renderNames(filter){const list=document.getElementById('namesList');let items=NAMES;if(filter){const q=filter.toLowerCase();items=items.filter(n=>n.n.toLowerCase().includes(q)||n.m.toLowerCase().includes(q)||n.ar.includes(filter));}else if(state.currentCat!=='all'){items=items.filter(n=>n.cat===state.currentCat);}if(items.length===0){list.innerHTML='<div class="empty"><svg viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><path d="M21 21l-4.35-4.35"/></svg><h3>No names found</h3><p>Try a different search</p></div>';return;}list.innerHTML=items.map((n,i)=>renderCard(n,i)).join('');}
function renderCard(n,i){const fav=state.favorites.includes(n.n);const genderClass=n.g==='male'?'male':'female';const genderLabel=n.g==='male'?'BOY':'GIRL';return '<div class="name-card" onclick="openDetail('+i+')"><div class="name-info"><div class="name-title">'+n.n+'<span class="name-gender '+genderClass+'">'+genderLabel+'</span></div><div class="name-meaning">'+n.m+'</div><div class="name-origin">'+n.o+'</div></div><div class="name-arabic">'+n.ar+'</div><button class="fav-btn'+(fav?' active':'')+'" onclick="event.stopPropagation();toggleFavByName(\''+n.n.replace(/'/g,"\\\\'")+'\')"><svg viewBox="0 0 24 24"><path d="M20.84 4.61a5.5 5.5 0 00-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 00-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 000-7.78z"/></svg></button></div>';}
function renderFavs(){const el=document.getElementById('favList');const favs=NAMES.filter(n=>state.favorites.includes(n.n));if(favs.length===0){el.innerHTML='<div class="empty"><svg viewBox="0 0 24 24"><path d="M20.84 4.61a5.5 5.5 0 00-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 00-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 000-7.78z"/></svg><h3>No favorites yet</h3><p>Tap the heart to save names</p></div>';return;}el.innerHTML=favs.map((n,i)=>renderCard(n,i)).join('');}
function toggleFavByName(name){const idx=state.favorites.indexOf(name);if(idx>-1){state.favorites.splice(idx,1);showToast('Removed');}else{state.favorites.push(name);showToast('Saved to favorites');}saveState();renderNames();renderFavs();}
function openDetail(i){const n=NAMES[i];if(!n)return;state.current=n;document.getElementById('modalArabic').textContent=n.ar;document.getElementById('modalName').textContent=n.n;document.getElementById('modalGender').textContent=(n.g==='male'?'Boy':'Girl')+' Name';document.getElementById('modalMeaning').textContent=n.m;document.getElementById('modalOrigin').textContent=n.o;const fav=state.favorites.includes(n.n);document.getElementById('modalFavBtn').classList.toggle('active-fav',fav);document.getElementById('modalFavText').textContent=fav?'Saved':'Favorite';document.getElementById('detailModal').classList.add('show');}
function closeModal(e){if(e&&e.target!==e.currentTarget)return;document.getElementById('detailModal').classList.remove('show');}
function toggleFavFromModal(){if(state.current){toggleFavByName(state.current.n);const fav=state.favorites.includes(state.current.n);document.getElementById('modalFavBtn').classList.toggle('active-fav',fav);document.getElementById('modalFavText').textContent=fav?'Saved':'Favorite';}}
function copyCurrent(){if(!state.current)return;const n=state.current;const text=n.n+' ('+n.ar+')\n'+n.m+'\n'+n.o;if(navigator.clipboard){navigator.clipboard.writeText(text).then(()=>showToast('Copied'));}}
function handleSearch(q){if(!q.trim()){renderNames();return;}renderNames(q);}
function switchView(view){document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));document.getElementById('view-'+view).classList.add('active');document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});const titles={names:'Islamic Names',fav:'Favorites',about:'About'};document.getElementById('headerTitle').textContent=titles[view]||'Islamic Names';document.getElementById('mainArea').scrollTo({top:0});}
let toastTimer;function showToast(msg){const t=document.getElementById('toast');document.getElementById('toastText').textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),1800);}
init();
