let state={theme:'dark',hijriMonth:null,hijriYear:null,gregMonth:null,gregYear:null,events:[]};
const HIJRI_MONTHS=['Muharram','Safar','Rabi al-Awwal','Rabi al-Thani','Jumada al-Ula','Jumada al-Akhirah','Rajab','Shaban','Ramadan','Shawwal','Dhul Qadah','Dhul Hijjah'];
const HIJRI_MONTHS_AR=['محرم','صفر','ربيع الأول','ربيع الثاني','جمادى الأولى','جمادى الآخرة','رجب','شعبان','رمضان','شوال','ذو القعدة','ذو الحجة'];
const EVENTS=[
{name:'Islamic New Year',hMonth:1,hDay:1,icon:'🌙'},
{name:'Day of Ashura',hMonth:1,hDay:10,icon:'💧'},
{name:'Mawlid an-Nabi ﷺ',hMonth:3,hDay:12,icon:'🕌'},
{name:'Isra & Mi\'raj',hMonth:7,hDay:27,icon:'✨'},
{name:'Mid-Sha\'ban',hMonth:8,hDay:15,icon:'🌕'},
{name:'Start of Ramadan',hMonth:9,hDay:1,icon:'🌙'},
{name:'Laylat al-Qadr (approx)',hMonth:9,hDay:27,icon:'⭐'},

{name:'Day of Arafah',hMonth:12,hDay:9,icon:'🏔️'},
{name:'Eid al-Adha',hMonth:12,hDay:10,icon:'🐑'}
];
function init(){loadState();applyTheme();loadCurrentHijriDate();}
function loadState(){try{const s=localStorage.getItem('hijri_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';}}catch(e){}}
function saveState(){try{localStorage.setItem('hijri_v1',JSON.stringify({theme:state.theme}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
async function loadCurrentHijriDate(){const now=new Date();const d=String(now.getDate()).padStart(2,'0');const m=String(now.getMonth()+1).padStart(2,'0');const y=now.getFullYear();try{const res=await fetch('https://api.aladhan.com/v1/gToH/'+d+'-'+m+'-'+y);const data=await res.json();if(data.data&&data.data.hijri){const h=data.data.hijri;state.hijriMonth=parseInt(h.month.number);state.hijriYear=parseInt(h.year);state.gregMonth=now.getMonth()+1;state.gregYear=now.getFullYear();renderCalendar();}else throw new Error('Invalid');}catch(err){console.error(err);state.hijriMonth=9;state.hijriYear=1446;state.gregMonth=now.getMonth()+1;state.gregYear=now.getFullYear();renderCalendar();}}
async function renderCalendar(){const now=new Date();const todayGreg={d:now.getDate(),m:now.getMonth()+1,y:now.getFullYear()};document.getElementById('calendarContent').innerHTML='<div class="loading"><div class="spinner"></div><p>Loading calendar...</p></div>';let todayHijri=null;try{const dd=String(todayGreg.d).padStart(2,'0');const mm=String(todayGreg.m).padStart(2,'0');const res=await fetch('https://api.aladhan.com/v1/gToH/'+dd+'-'+mm+'-'+todayGreg.y);const data=await res.json();if(data.data&&data.data.hijri){todayHijri={day:parseInt(data.data.hijri.day),month:parseInt(data.data.hijri.month.number),year:parseInt(data.data.hijri.year)};}}catch(e){todayHijri={day:1,month:9,year:1446};}
let html='';
html+='<div class="hero"><div class="hero-content"><div class="hero-label">Today\\'s Date</div><div class="hero-day">'+todayHijri.day+'</div><div class="hero-month">'+HIJRI_MONTHS_AR[todayHijri.month-1]+'</div><div class="hero-year">'+HIJRI_MONTHS[todayHijri.month-1]+' '+todayHijri.year+' AH</div><div class="hero-gregorian">📅 '+now.toLocaleDateString('en-PK',{weekday:'long',day:'numeric',month:'long',year:'numeric'})+'</div></div></div>';
html+='<div class="month-head"><button class="month-nav" onclick="changeMonth(-1)"><svg viewBox="0 0 24 24"><path d="M15 18l-6-6 6-6"/></svg></button><div class="month-label"><div class="month-name-ar">'+HIJRI_MONTHS_AR[state.hijriMonth-1]+'</div><div class="month-name-en">'+HIJRI_MONTHS[state.hijriMonth-1]+' '+state.hijriYear+' AH</div></div><button class="month-nav" onclick="changeMonth(1)"><svg viewBox="0 0 24 24"><path d="M9 18l6-6-6-6"/></svg></button></div>';
html+='<div class="weekdays"><div class="weekday">Sun</div><div class="weekday">Mon</div><div class="weekday">Tue</div><div class="weekday">Wed</div><div class="weekday">Thu</div><div class="weekday">Fri</div><div class="weekday">Sat</div></div>';
html+='<div class="days-grid" id="daysGrid"></div>';
html+='<div class="events-title">Events this month</div>';
html+='<div id="monthEvents"></div>';
document.getElementById('calendarContent').innerHTML=html;
renderDaysGrid(todayHijri);renderMonthEvents(todayHijri);}
function changeMonth(delta){state.hijriMonth+=delta;if(state.hijriMonth>12){state.hijriMonth=1;state.hijriYear++;}if(state.hijriMonth<1){state.hijriMonth=12;state.hijriYear--;}const now=new Date();renderCalendar();}
async function renderDaysGrid(todayHijri){const grid=document.getElementById('daysGrid');grid.innerHTML='';const daysInMonth=getHijriMonthDays(state.hijriMonth,state.hijriYear);const firstDayOfMonth=await getFirstDayOfWeek(state.hijriMonth,state.hijriYear);for(let i=0;i<firstDayOfMonth;i++){const empty=document.createElement('div');empty.className='day empty';grid.appendChild(empty);}const eventDays=EVENTS.filter(e=>e.hMonth===state.hijriMonth).map(e=>e.hDay);for(let d=1;d<=daysInMonth;d++){const dayEl=document.createElement('div');const isToday=state.hijriMonth===todayHijri.month&&state.hijriYear===todayHijri.year&&d===todayHijri.day;const hasEvent=eventDays.includes(d);dayEl.className='day'+(isToday?' today':'')+(hasEvent?' event':'');dayEl.innerHTML='<span>'+d+'</span>'+(hasEvent?'<div class="day-dot"></div>':'');dayEl.onclick=()=>showDayInfo(d);grid.appendChild(dayEl);}}
function getHijriMonthDays(month,year){if(month%2===1)return 30;if(month===12)return isLeapYear(year)?30:29;return 29;}
function isLeapYear(year){return (year*11+14)%30<11;}
async function getFirstDayOfWeek(hMonth,hYear){try{const res=await fetch('https://api.aladhan.com/v1/hToG/'+hMonth+'/'+hYear);const data=await res.json();if(data.data&&data.data.gregorian){const g=data.data.gregorian;const d=new Date(g.year+'-'+String(g.month.number).padStart(2,'0')+'-'+String(g.day).padStart(2,'0'));return d.getDay();}}catch(e){}return 0;}
function renderMonthEvents(todayHijri){const container=document.getElementById('monthEvents');const monthEvents=EVENTS.filter(e=>e.hMonth===state.hijriMonth);if(monthEvents.length===0){container.innerHTML='<div style="text-align:center;padding:20px;color:var(--text-subtle);font-size:13px">No major events this month</div>';return;}container.innerHTML=monthEvents.map(e=>'<div class="event-item"><div class="event-icon">'+e.icon+'</div><div class="event-info"><div class="event-name">'+e.name+'</div><div class="event-date">'+e.hDay+' '+HIJRI_MONTHS[state.hijriMonth-1]+' '+state.hijriYear+' AH</div></div></div>').join('');}
function renderEventsView(){const container=document.getElementById('eventsContent');container.innerHTML=EVENTS.map(e=>'<div class="event-item"><div class="event-icon">'+e.icon+'</div><div class="event-info"><div class="event-name">'+e.name+'</div><div class="event-date">'+e.hDay+' '+HIJRI_MONTHS[e.hMonth-1]+'</div></div></div>').join('');}
async function convertToHijri(){const input=document.getElementById('gregInput');if(!input.value){showToast('Select a date');return;}const d=new Date(input.value);const dd=String(d.getDate()).padStart(2,'0');const mm=String(d.getMonth()+1).padStart(2,'0');const yyyy=d.getFullYear();try{const res=await fetch('https://api.aladhan.com/v1/gToH/'+dd+'-'+mm+'-'+yyyy);const data=await res.json();if(data.data&&data.data.hijri){const h=data.data.hijri;document.getElementById('hijriResultVal').textContent=h.day+' '+h.month.en+' '+h.year+' AH';document.getElementById('hijriResultSub').textContent=h.month.ar+' · '+h.weekday.en;document.getElementById('gregResult').style.display='block';}}catch(err){showToast('Conversion failed');}}
async function convertToGreg(){const day=parseInt(document.getElementById('hDay').value);const month=parseInt(document.getElementById('hMonth').value);const year=parseInt(document.getElementById('hYear').value);if(!day||day<1||day>30){showToast('Enter valid day (1-30)');return;}if(!year||year<1){showToast('Enter valid year');return;}try{const res=await fetch('https://api.aladhan.com/v1/hToG/'+day+'-'+month+'-'+year);const data=await res.json();if(data.data&&data.data.gregorian){const g=data.data.gregorian;document.getElementById('gregResultVal').textContent=g.day+' '+g.month.en+' '+g.year+' CE';document.getElementById('gregResultSub').textContent=g.weekday.en+' · '+g.designation;document.getElementById('hijriResult').style.display='block';}}catch(err){showToast('Conversion failed');}}
function showDayInfo(d){const events=EVENTS.filter(e=>e.hMonth===state.hijriMonth&&e.hDay===d);if(events.length>0){showToast(events[0].icon+' '+events[0].name);}else{showToast(d+' '+HIJRI_MONTHS[state.hijriMonth-1]+' '+state.hijriYear+' AH');}}
function switchView(view){document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));document.getElementById('view-'+view).classList.add('active');document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});const titles={calendar:'Hijri Calendar',events:'Islamic Events',converter:'Date Converter'};document.getElementById('headerTitle').textContent=titles[view]||'Calendar';document.getElementById('mainArea').scrollTo({top:0});if(view==='events')renderEventsView();}
let toastTimer;function showToast(msg){const t=document.getElementById('toast');document.getElementById('toastText').textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),2000);}
