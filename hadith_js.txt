let state={theme:'dark',fontSize:'medium',currentCat:'all',favorites:[],current:null};
function init(){loadState();applyTheme();applyFontSize();renderCategories();renderHadiths();renderFavs();}
function loadState(){try{const s=localStorage.getItem('hadith_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.fontSize=p.fontSize||'medium';state.favorites=p.favorites||[];}}catch(e){}}
function saveState(){try{localStorage.setItem('hadith_v1',JSON.stringify({theme:state.theme,fontSize:state.fontSize,favorites:state.favorites}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
function applyFontSize(){const sizes={small:'20px',medium:'24px',large:'30px'};document.documentElement.style.setProperty('--font-arabic',sizes[state.fontSize]);document.querySelectorAll('.font-btn').forEach(b=>{b.classList.toggle('active',b.dataset.size===state.fontSize);});}
function setFontSize(size){state.fontSize=size;applyFontSize();saveState();}
function renderCategories(){const el=document.getElementById('catChips');el.innerHTML=CATEGORIES.map(c=>'<button class="chip'+(c.id===state.currentCat?' active':'')+'" onclick="setCategory(\''+c.id+'\')">'+c.icon+' '+c.name+'</button>').join('');}
function setCategory(id){state.currentCat=id;renderCategories();renderHadiths();document.getElementById('mainArea').scrollTo({top:0,behavior:'smooth'});}
function renderHadiths(filter){const list=document.getElementById('hadithList');let items=HADITHS;if(filter){const q=filter.toLowerCase();items=items.filter(h=>h.en.toLowerCase().includes(q)||h.ar.includes(filter)||h.ref.toLowerCase().includes(q));}else if(state.currentCat!=='all'){items=items.filter(h=>h.cat===state.currentCat);}if(items.length===0){list.innerHTML='<div class="empty"><h3>No hadith found</h3><p>Try a different search</p></div>';return;}list.innerHTML=items.map(h=>renderCard(h)).join('');}
function renderCard(h){const fav=state.favorites.includes(h.id);const cat=CATEGORIES.find(c=>c.id===h.cat)||{name:h.cat};const shortEn=h.en.length>100?h.en.slice(0,100)+'...':h.en;return '<div class="hadith-card" onclick="openDetail('+h.id+')"><div class="hadith-head"><span class="hadith-cat">'+cat.name+'</span><button class="fav-btn'+(fav?' active':'')+'" onclick="event.stopPropagation();toggleFav('+h.id+')"><svg viewBox="0 0 24 24"><path d="M20.84 4.61a5.5 5.5 0 00-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 00-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 000-7.78z"/></svg></button></div><div class="hadith-arabic">'+h.ar+'</div><div class="hadith-english">'+shortEn+'</div><div class="hadith-ref">'+h.ref+'</div></div>';}
function renderFavs(){const el=document.getElementById('favList');const favs=HADITHS.filter(h=>state.favorites.includes(h.id));if(favs.length===0){el.innerHTML='<div class="empty"><svg viewBox="0 0 24 24"><path d="M20.84 4.61a5.5 5.5 0 00-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 00-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 000-7.78z"/></svg><h3>No favorites yet</h3><p>Tap the heart to save</p></div>';return;}el.innerHTML=favs.map(h=>renderCard(h)).join('');}
function toggleFav(id){const idx=state.favorites.indexOf(id);if(idx>-1){state.favorites.splice(idx,1);showToast('Removed');}else{state.favorites.push(id);showToast('Saved');}saveState();renderHadiths();renderFavs();}
function openDetail(id){const h=HADITHS.find(x=>x.id===id);if(!h)return;state.current=h;const cat=CATEGORIES.find(c=>c.id===h.cat)||{name:h.cat};document.getElementById('modalCat').textContent=cat.name;document.getElementById('modalArabic').textContent=h.ar;document.getElementById('modalEnglish').textContent=h.en;document.getElementById('modalRef').textContent='📚 '+h.ref;const fav=state.favorites.includes(h.id);document.getElementById('modalFavBtn').classList.toggle('active-fav',fav);document.getElementById('modalFavText').textContent=fav?'Saved':'Favorite';document.getElementById('detailModal').classList.add('show');}
function closeModal(e){if(e&&e.target!==e.currentTarget)return;document.getElementById('detailModal').classList.remove('show');}
function toggleFavFromModal(){if(state.current){toggleFav(state.current.id);const fav=state.favorites.includes(state.current.id);document.getElementById('modalFavBtn').classList.toggle('active-fav',fav);document.getElementById('modalFavText').textContent=fav?'Saved':'Favorite';}}
function copyCurrent(){if(!state.current)return;const h=state.current;const text=h.ar+'\n\n'+h.en+'\n\n— '+h.ref;if(navigator.clipboard){navigator.clipboard.writeText(text).then(()=>showToast('Copied'));}}
function handleSearch(q){if(!q.trim()){renderHadiths();return;}renderHadiths(q);}
function switchView(view){document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));document.getElementById('view-'+view).classList.add('active');document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});const titles={all:'Hadith Collection',fav:'Favorites',settings:'Settings'};document.getElementById('headerTitle').textContent=titles[view]||'Hadith';document.getElementById('mainArea').scrollTo({top:0});}
let toastTimer;function showToast(msg){const t=document.getElementById('toast');document.getElementById('toastText').textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),1800);}
init();
