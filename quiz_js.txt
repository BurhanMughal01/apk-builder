let state={theme:'dark',currentCat:null,currentQ:0,questions:[],score:0,answered:false,timeLeft:30,timer:null,highScores:[],correct:0,wrong:0};
function init(){loadState();applyTheme();renderStart();}
function loadState(){try{const s=localStorage.getItem('quiz_v1');if(s){const p=JSON.parse(s);state.theme=p.theme||'dark';state.highScores=p.highScores||[];}}catch(e){}}
function saveState(){try{localStorage.setItem('quiz_v1',JSON.stringify({theme:state.theme,highScores:state.highScores}));}catch(e){}}
function applyTheme(){document.body.setAttribute('data-theme',state.theme);const icon=document.getElementById('themeIcon');if(icon){if(state.theme==='dark'){icon.innerHTML='<circle cx="12" cy="12" r="5"/><path d="M12 1v2M12 21v2M4.22 4.22l1.42 1.42M18.36 18.36l1.42 1.42M1 12h2M21 12h2M4.22 19.78l1.42-1.42M18.36 5.64l1.42-1.42"/>';}else{icon.innerHTML='<path d="M21 12.79A9 9 0 1111.21 3 7 7 0 0021 12.79z"/>';}}}
function toggleTheme(){state.theme=state.theme==='dark'?'light':'dark';applyTheme();saveState();}
function renderStart(){state.currentCat=null;state.questions=[];state.currentQ=0;state.score=0;state.correct=0;state.wrong=0;if(state.timer){clearInterval(state.timer);state.timer=null;}const html='<div class="hero"><div class="hero-icon">🎮</div><div class="hero-title">Islamic Quiz</div><div class="hero-sub">Test your knowledge</div></div><div class="start-screen">'+CATEGORIES.map(c=>'<button class="category-btn" onclick="startQuiz(\''+c.id+'\')"><div class="cat-icon">'+c.icon+'</div><div class="cat-info"><div class="cat-name">'+c.name+'</div><div class="cat-desc">'+c.desc+'</div></div><svg class="cat-arrow" viewBox="0 0 24 24"><path d="M9 18l6-6-6-6"/></svg></button>').join('')+'</div>';document.getElementById('quizContent').innerHTML=html;}
function startQuiz(cat){state.currentCat=cat;state.questions=[...QUESTIONS[cat]].sort(()=>Math.random()-0.5);state.currentQ=0;state.score=0;state.correct=0;state.wrong=0;state.answered=false;renderQuestion();}
function renderQuestion(){if(state.currentQ>=state.questions.length){renderResult();return;}const q=state.questions[state.currentQ];state.answered=false;state.timeLeft=30;const letters=['A','B','C','D'];const progress=((state.currentQ)/state.questions.length)*100;let html='<div class="quiz-screen">';html+='<div class="quiz-head"><div><div class="quiz-progress-text">Question '+(state.currentQ+1)+'/'+state.questions.length+'</div></div><div style="display:flex;align-items:center;gap:10px"><span class="quiz-score-live">⭐ '+state.score+'</span><div class="timer-badge" id="timerBadge"><svg viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M12 6v6l4 2"/></svg><span id="timerText">30s</span></div></div></div>';html+='<div class="quiz-progress-bar"><div class="quiz-progress-fill" style="width:'+progress+'%"></div></div>';html+='<div class="question-card"><div class="q-num">Question '+(state.currentQ+1)+'</div><div class="q-text">'+q.q+'</div></div>';html+='<div class="answers-list" id="answersList">';q.a.forEach((ans,i)=>{html+='<button class="answer-btn" id="ans-'+i+'" onclick="selectAnswer('+i+')"><span class="answer-letter">'+letters[i]+'</span><span>'+ans+'</span></button>';});html+='</div></div>';document.getElementById('quizContent').innerHTML=html;startTimer();}
function startTimer(){if(state.timer)clearInterval(state.timer);const tb=document.getElementById('timerBadge');const tt=document.getElementById('timerText');state.timer=setInterval(()=>{state.timeLeft--;if(tt)tt.textContent=state.timeLeft+'s';if(tb){tb.classList.toggle('warning',state.timeLeft<=10);}if(state.timeLeft<=0){clearInterval(state.timer);state.timer=null;if(!state.answered)timeUp();}},1000);}
function timeUp(){state.answered=true;state.wrong++;const q=state.questions[state.currentQ];const correctBtn=document.getElementById('ans-'+q.c);if(correctBtn)correctBtn.classList.add('reveal-correct');document.querySelectorAll('.answer-btn').forEach(b=>b.disabled=true);showToast('Time up!');setTimeout(()=>{state.currentQ++;renderQuestion();},1500);}
function selectAnswer(idx){if(state.answered)return;state.answered=true;if(state.timer){clearInterval(state.timer);state.timer=null;}const q=state.questions[state.currentQ];const correct=q.c;if(idx===correct){state.score+=10;state.correct++;if(navigator.vibrate)navigator.vibrate(50);document.getElementById('ans-'+idx).classList.add('correct');showToast('Correct! +10');}else{state.wrong++;if(navigator.vibrate)navigator.vibrate([50,30,50]);document.getElementById('ans-'+idx).classList.add('wrong');const cb=document.getElementById('ans-'+correct);if(cb)cb.classList.add('reveal-correct');}document.querySelectorAll('.answer-btn').forEach(b=>b.disabled=true);setTimeout(()=>{state.currentQ++;renderQuestion();},1500);}
function renderResult(){const total=state.questions.length;const percent=Math.round((state.correct/total)*100);let title='Keep Learning!';let msg='Practice makes perfect.';if(percent>=90){title='MashaAllah! 🎉';msg='Outstanding performance!';}else if(percent>=70){title='Excellent! 🌟';msg='You have great knowledge!';}else if(percent>=50){title='Good Job! 👍';msg='Keep improving!';}else{title='Keep Learning! 📚';msg='Practice makes perfect.';}state.highScores.unshift({cat:state.currentCat,total:total,correct:state.correct,score:state.score,percent:percent,date:new Date().toISOString()});if(state.highScores.length>20)state.highScores=state.highScores.slice(0,20);saveState();let html='<div class="result-screen">';html+='<div class="result-circle"><div class="result-percent">'+percent+'%</div><div class="result-percent-label">SCORE</div></div>';html+='<div class="result-title">'+title+'</div>';html+='<div class="result-message">'+msg+'</div>';html+='<div class="result-stats"><div class="result-stat"><div class="result-stat-val">'+state.correct+'</div><div class="result-stat-label">Correct</div></div><div class="result-stat"><div class="result-stat-val">'+state.wrong+'</div><div class="result-stat-label">Wrong</div></div><div class="result-stat"><div class="result-stat-val">'+state.score+'</div><div class="result-stat-label">Points</div></div></div>';html+='<div class="result-actions"><button class="btn-primary" onclick="startQuiz(\''+state.currentCat+'\')">🔄 Try Again</button><button class="btn-secondary" onclick="renderStart()">🏠 Choose Category</button></div></div>';document.getElementById('quizContent').innerHTML=html;}
function renderScores(){const el=document.getElementById('scoresList');if(!state.highScores||state.highScores.length===0){el.innerHTML='<div class="empty"><svg viewBox="0 0 24 24"><path d="M8 21h8M12 17v4M7 4h10l1 8a6 6 0 11-12 0z"/></svg><h3>No scores yet</h3><p>Take a quiz to see your results</p></div>';return;}el.innerHTML=state.highScores.map((s,i)=>{const cat=CATEGORIES.find(c=>c.id===s.cat)||{name:s.cat,icon:'🎮'};const date=new Date(s.date);const dateStr=date.toLocaleDateString('en-PK',{day:'numeric',month:'short'})+' · '+date.toLocaleTimeString('en-PK',{hour:'2-digit',minute:'2-digit'});return '<div class="lb-item"><div class="lb-rank">'+(i+1)+'</div><div class="lb-info"><div class="lb-cat">'+cat.icon+' '+cat.name+'</div><div class="lb-date">'+dateStr+'</div></div><div class="lb-score">'+s.score+'</div></div>';}).join('');}
function switchView(view){document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));document.getElementById('view-'+view).classList.add('active');document.querySelectorAll('.nav-item').forEach(b=>{b.classList.toggle('active',b.dataset.view===view);});if(view==='scores')renderScores();document.getElementById('headerTitle').textContent=view==='scores'?'Your Scores':'Islamic Quiz';document.getElementById('mainArea').scrollTo({top:0});}
let toastTimer;function showToast(msg){const t=document.getElementById('toast');document.getElementById('toastText').textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),1500);}
init();
